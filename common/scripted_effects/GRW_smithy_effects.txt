GRW_update_smithy_display_vars_effect = {
	set_variable = { GRW_smithy_disp_prod = 0 }
	set_variable = { GRW_smithy_disp_con = 0 }
	set_variable = { GRW_smithy_disp_cg_factor = 0 }
	set_variable = { GRW_smithy_disp_cg_expected = 0 }
	set_variable = { GRW_smithy_disp_eff = 0 }

	set_variable = { GRW_smithy_level = GRW_smithy_scale }
	if = {
		limit = { check_variable = { GRW_smithy_scale < 0 } }
		multiply_variable = { GRW_smithy_level = -1 }
	}

	if = {
		limit = { check_variable = { GRW_smithy_scale > 0 } }
		set_variable = { GRW_smithy_disp_prod = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_prod = 0.035 }

		set_variable = { GRW_smithy_disp_con = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_con = -0.05 }

		set_variable = { GRW_smithy_disp_cg_expected = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_cg_expected = 0.02 }

		set_variable = { GRW_smithy_disp_eff = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_eff = -0.05 }
	}
	else_if = {
		limit = { check_variable = { GRW_smithy_scale < 0 } }
		set_variable = { GRW_smithy_disp_prod = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_prod = 0.03 }

		set_variable = { GRW_smithy_disp_con = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_con = -0.05 }

		set_variable = { GRW_smithy_disp_cg_factor = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_cg_factor = 0.04 }

		set_variable = { GRW_smithy_disp_eff = GRW_smithy_scale }
		multiply_variable = { GRW_smithy_disp_eff = 0.05 }
	}
}

GRW_smithy_refresh_effect = {
	GRW_update_smithy_ideas_effect = yes
	GRW_update_smithy_display_vars_effect = yes
}
